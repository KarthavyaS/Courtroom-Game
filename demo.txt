# 4. BUTTON CLASS
# ==========================================

class Button:

    def __init__(self, x, y, width, height, text):

        self.rect = pygame.Rect(x, y, width, height)
        self.text = text

    def draw(self, surface):

        mouse = pygame.mouse.get_pos()

        color = GOLD if self.rect.collidepoint(mouse) else WOOD

        pygame.draw.rect(
            surface, color, self.rect, border_radius=10
        )

        pygame.draw.rect(
            surface, GOLD, self.rect, 2, border_radius=10
        )

        text_surface = button_font.render(
            self.text, True, LIGHT_PARCHMENT
        )

        surface.blit(
            text_surface,
            text_surface.get_rect(center=self.rect.center)
        )

    def is_clicked(self, position):

        return self.rect.collidepoint(position)

# 6. CASE CARD CLASS
# ==========================================

class CaseCard:

    def __init__(self, x, y, width, height, case):

        self.rect = pygame.Rect(x, y, width, height)
        self.case = case

        self.investigate_rect = pygame.Rect(
            x + 25, y + height - 65, width - 50, 43
        )

    def draw(self, surface, selected=False):

        mouse = pygame.mouse.get_pos()

        hovering = self.rect.collidepoint(mouse)

        # Card background
        card_color = (
            LIGHT_PARCHMENT if hovering or selected
            else CARD_COLOR
        )

        pygame.draw.rect(
            surface, card_color, self.rect, border_radius=15
        )

        # Card border
        border_color = GOLD if selected or hovering else WOOD

        pygame.draw.rect(
            surface, border_color, self.rect, 3,
            border_radius=15
        )

        # Case ID
        case_id = small_font.render(
            self.case["id"], True, MUTED
        )

        surface.blit(case_id, (self.rect.x + 22, self.rect.y + 18))

        # Case title
        title = heading_font.render(
            self.case["title"], True, DARK_BROWN
        )

        # Smaller title font for long titles
        if title.get_width() > self.rect.width - 35:
            title = button_font.render(
                self.case["title"], True, DARK_BROWN
            )

        surface.blit(
            title, (self.rect.x + 20, self.rect.y + 55)
        )

        # Case type
        case_type = normal_font.render(
            self.case["type"], True, MUTED
        )

        surface.blit(
            case_type, (self.rect.x + 22, self.rect.y + 105)
        )

        # Description
        words = self.case["description"].split()
        lines = []
        current_line = ""

        for word in words:

            test_line = current_line + word + " "

            if normal_font.size(test_line)[0] < self.rect.width - 45:
                current_line = test_line
            else:
                lines.append(current_line)
                current_line = word + " "

        if current_line:
            lines.append(current_line)

        for i, line in enumerate(lines):

            text = normal_font.render(
                line, True, DARK_BROWN
            )

            surface.blit(
                text,
                (self.rect.x + 22, self.rect.y + 145 + i * 25)
            )

        # Difficulty
        difficulty = small_font.render(
            "Difficulty: " + self.case["difficulty"],
            True, DARK_BROWN
        )

        surface.blit(
            difficulty, (self.rect.x + 22, self.rect.y + 230)
        )

        # Suspects and evidence
        details = small_font.render(
            "Suspects: " + str(self.case["suspects"]) +
            "    Evidence: " + str(self.case["evidence"]),
            True, DARK_BROWN
        )

        surface.blit(
            details, (self.rect.x + 22, self.rect.y + 265)
        )

        # Reward
        reward = normal_font.render(
            "Reward: " + str(self.case["reward"]) + " points",
            True, DARK_BROWN
        )

        surface.blit(
            reward, (self.rect.x + 22, self.rect.y + 295)
        )

        # Investigate button
        button_color = GOLD if self.investigate_rect.collidepoint(mouse) else WOOD

        pygame.draw.rect(
            surface, button_color,
            self.investigate_rect,
            border_radius=8
        )

        button_text = button_font.render(
            "INVESTIGATE", True, WHITE
        )

        surface.blit(
            button_text,
            button_text.get_rect(center=self.investigate_rect.center)
        )

    def clicked(self, position):

        return self.rect.collidepoint(position)

    def investigate_clicked(self, position):

        return self.investigate_rect.collidepoint(position)

        True, LIGHT_PARCHMENT

# 10. CASE SELECTION SCREEN

def draw_case_selection():

    screen.fill(BACKGROUND)

    # Page border
    pygame.draw.rect(
        screen, DARK_BROWN,
        (15, 15, WIDTH - 30, HEIGHT - 30),
        4, border_radius=15

    heading = title_font.render(

        heading,
        heading.get_rect(center=(WIDTH // 2, 100))

    subtitle = normal_font.render(

        True, MUTED
    )

    screen.blit(
        subtitle,

    back_button.draw(screen)

    # Draw all cards
    for i, card in enumerate(case_cards):

        card.draw(
            screen,
            selected=(selected_case == i)
        )


# ==========================================
# 11. PLACEHOLDER SCREEN

def draw_placeholder(title):

    screen.fill(BACKGROUND)

    heading = title_font.render(
        title, True, DARK_BROWN
    )

    screen.blit(
        heading,
        heading.get_rect(center=(WIDTH // 2, 280))
    )

    message = normal_font.render(
        "This module will be developed next.",
        True, MUTED
    )

    screen.blit(
        message,
        message.get_rect(center=(WIDTH // 2, 370))
    )

    back_button.draw(screen)


# ==========================================
# 12. GAME LOOP
# ==========================================

running = True
current_screen = "menu"

while running:

    for event in pygame.event.get():

        if event.type == pygame.QUIT:
            running = False

        if event.type == pygame.MOUSEBUTTONDOWN:

            if event.button == 1:

                if current_screen == "menu":

                    if start_button.is_clicked(event.pos):
                        current_screen = "cases"

                    elif create_button.is_clicked(event.pos):
                        current_screen = "create"

                    elif leaderboard_button.is_clicked(event.pos):
                        current_screen = "leaderboard"

                    elif exit_button.is_clicked(event.pos):
                        running = False

                elif current_screen == "cases":

                    if back_button.is_clicked(event.pos):

                        current_screen = "menu"

                    else:

                        for i, card in enumerate(case_cards):

                            if card.clicked(event.pos):
                                selected_case = i

                            if card.investigate_clicked(event.pos):

                                selected_case = i

                                if i == 0:
                                    current_screen = "crime_scene"

                                else:
                                    current_screen = "case_details"

                else:

                    if back_button.is_clicked(event.pos):
                        current_screen = "cases"

    # DRAW SCREENS

    if current_screen == "menu":

        draw_main_menu()

    elif current_screen == "cases":

        draw_case_selection()

    elif current_screen == "crime_scene":

        draw_placeholder("CRIME SCENE")

    elif current_screen == "case_details":

        draw_placeholder("CASE DETAILS")

    elif current_screen == "create":

        draw_placeholder("CREATE CASE")

    elif current_screen == "leaderboard":

        draw_placeholder("LEADERBOARD")

    pygame.display.flip()

    clock.tick(60)


# ==========================================
# 13. QUIT
# ==========================================

pygame.quit()
sys.exit()
